\begin{frame}{Модель системы и ключевая идея}
        \footnotesize
        \begin{columns}[T]
                \column{0.5\textwidth}
                \textbf{Аппаратная платформа}
                \begin{itemize}
                        \item Ядра: $\mathcal{P} = \{P_1,\ldots,P_{K_P}\}$,\;
                                $\mathcal{E} = \{E_1,\ldots,E_{K_E}\}$
                        \item Дискретное время $t = 1,2,3,\ldots$
                        \item Замедление на E-ядре:
                                $\sigma = \eta_P/\eta_E > 1$
                        \item Общий ISA (любая задача может быть
                                выполнена на любом ядре)
                \end{itemize}

                \column{0.5\textwidth}
                \textbf{Задачи как агенты}
                \begin{itemize}
                        \item Тип $\theta_i = (v_i, l_i)$: ценность $v_i$,
                                длина $l_i$
                        \item Полезность как маргинальный вклад в
                                социальное благосостояние $W$:
                                $U_i = W(s^t) - W_{-i}(s^t) \geq 0$
                        \item Бюджет $B_i(\chi_i) > 0$ по классу
                                обслуживания $\chi_i$:
                \end{itemize}
                \vspace{0.2em}
                \hspace{2em}{\footnotesize
                        \begin{tabular}{@{}l@{}}
                                реального времени \\
                                интерактивный \\
                                пакетный \\
                                фоновый
                \end{tabular}}
        \end{columns}

        \vspace{0.5em}
        \begin{shadowbox4}
                \scriptsize Временные кванты P/E\nobreakdash-ядер
                процессора~-- \textbf{недолговечный ресурс}
                (неиспользованный квант теряется безвозвратно). Их
                распределение моделируется динамическим аукционом VCG
                с задачами\nobreakdash-агентами, максимизирующим социальное
                благосостояние
        \end{shadowbox4}
\end{frame}


\begin{frame}{MDP-модель и механизм VCG}
        \footnotesize
        \begin{columns}[T]
                \column{0.5\textwidth}
                \textbf{Состояние} $s^t =
                (\mathbf{x}^t,\,\mathbf{b}^t,\,\Theta^t,\,\mathcal{N}^t)$
                {\scriptsize
                \begin{itemize}\setlength{\itemsep}{0pt}
                        \item $\mathbf{x}^t$~-- остаточные длительности
                                контрактов
                        \item $\mathbf{b}^t$~-- остаточные бюджеты
                        \item $\Theta^t$~-- типы задач,
                                $\mathcal{N}^t$~-- активные задачи
                \end{itemize}}

                \vspace{0.2em}
                \textbf{Действие} $a_i^t = (a_{i,P}, a_{i,E})
                \in \{0,1\}^2$, $a_{i,P} + a_{i,E} \leq 1$
                $$\sum_i a_{i,P} \leq K_P^t,\quad
                \sum_i a_{i,E} \leq K_E^t$$
                {\scriptsize ($K_\kappa^t$~-- число свободных ядер
                кластера $\kappa$ в момент $t$)}

                \vspace{0.1em}
                \textbf{Эффективная ценность} ($\kappa \in \{P, E\}$)
                $$\phi_\kappa(\theta_i) = v_i - c_\kappa\, m_\kappa(l_i)$$
                {\scriptsize $m_P(l) = l$,\; $m_E(l) = \lceil l\sigma \rceil$}

                \column{0.5\textwidth}
                \textbf{Социальное благосостояние} (Беллман,
                $\delta \in (0,1)$):
                $$W(s^t) = \max_{a^t} \Bigl[\,
                {\textstyle\sum_i} \phi^{a}_i
                + \delta\, \mathbb{E}[W(s^{t+1})]\Bigr]$$
                {\scriptsize $\phi^{a}_i = a_{i,P}\phi_P(\theta_i)
                + a_{i,E}\phi_E(\theta_i)$}

                \vspace{4em}
                \textbf{Платёж VCG}
                $$p_i^t = W_{-i}(s^t) - \bigl(W(s^t) -
                \phi_\kappa(\theta_i)\bigr)$$

        \end{columns}
\end{frame}


\begin{frame}{Реализация}
        \footnotesize
        \textbf{Решение аукциона}

        Задача направляется в DSQ\_P или DSQ\_E по
        $\kappa^* = \arg\max_{\kappa\in\{P,E\}} \phi_\kappa(\theta_i)$

        \vspace{0.3em}
        \begin{columns}[T]
                \column{0.33\textwidth}
                \begin{shadowbox2}
                        \textbf{DSQ\_P}

                        \vspace{0.2em}
                        \footnotesize
                        $\phi_P \geq \phi_E$

                        \vspace{0.1em}
                        P-ядро выгоднее по эффективной ценности
                \end{shadowbox2}

                \column{0.33\textwidth}
                \begin{shadowbox2}
                        \textbf{DSQ\_E}

                        \vspace{0.2em}
                        \footnotesize
                        $\phi_E > \phi_P$ либо короткая задача
                        при перегрузке P-кластера

                        \vspace{0.1em}
                        {\scriptsize $Q_P \geq K_P$ и $Q_P K_E > Q_E K_P$}
                \end{shadowbox2}

                \column{0.33\textwidth}
                \begin{shadowbox2}
                        \textbf{DSQ\_Starved}

                        \vspace{0.2em}
                        \footnotesize
                        $B_i^t < B_i^{\max} / 10$

                        \vspace{0.1em}
                        Низкий приоритет, обслуживается после пополнения
                \end{shadowbox2}
        \end{columns}

        \vspace{0.6em}
        \footnotesize\textbf{Ослабления модели в реализации}

        \vspace{0.2em}
        \begin{columns}[T]
                \column{0.5\textwidth}
                {\footnotesize
                \begin{itemize}\setlength{\itemsep}{0pt}
                        \item $v_i$ сообщается через вес задачи
                                \texttt{p->scx.weight}
                        \item $W_{-i}(s^t) \to$ кластерная EWMA
                                $\bar W_\kappa$ по реализованным
                                $|\phi|$
                        \item $l_i \to$ EWMA по истории исполнения
                \end{itemize}}

                \column{0.5\textwidth}
                {\footnotesize
                \begin{itemize}\setlength{\itemsep}{0pt}
                        \item Контракт~-- горизонт планирования,
                                не атомарная единица
                        \item Бюджет~-- корзина токенов с
                                пополнением во сне
                                ($B_i^{\max} = w_i\cdot\texttt{BUDGET\_MUL}$,
                                $\Delta B_i \propto w_i$)
                \end{itemize}}
        \end{columns}
\end{frame}

\begin{frame}{Extra. Глоссарий}
        \scriptsize
        \begin{description}
                \item[EEVDF] Earliest Eligible Virtual Deadline First~--- штатный
                        планировщик ядра Linux с 6.6
                \item[sched\_ext] инфраструктура ядра для подключаемых
                        планировщиков на eBPF
                \item[eBPF] extended Berkeley Packet Filter, верифицируемый
                        байт-код, исполняемый в ядре
                \item[ghOSt] пользовательский планировщик от Google,
                        политика в user-space
                \item[scx\_LAVD] Latency-Aware Virtual Deadline, eBPF-планировщик
                        с учётом отзывчивости
                \item[scx\_EEVDF] порт EEVDF поверх sched\_ext, эталон накладных
                        расходов фреймворка
                \item[DSQ] Dispatch Queue, очередь готовых задач внутри sched\_ext
                \item[VCG] Vickrey-Clarke-Groves, аукцион с правдивыми оценками
                \item[MDP] Markov Decision Process, марковский процесс принятия
                        решений
                \item[EWMA] Exponentially Weighted Moving Average,
                        экспоненциально-взвешенное скользящее среднее
                \item[RAPL] Running Average Power Limit, счётчики энергии Intel
                \item[p99 / p99.9] 99-й и 99{,}9-й процентили задержки
                \item[P / E ядра] Performance / Efficient ядра в гибридной
                        архитектуре Intel
                \item[$\sigma$] коэффициент замедления, отношение пиковой
                        производительности P-ядра к E-ядру
        \end{description}
\end{frame}

\begin{frame}{Extra. Пример решения аукциона}
        \textbf{Пример: $K_P = K_E = 1$, одно свободное ядро типа $\kappa$}

        \vspace{0.2cm}
        \textbf{Победитель:}
        $$i^* = \arg\max_{i \in \mathcal{N}^t,\; B_i^t > 0}\;
        \phi_\kappa(\theta_i) + \delta^{m_\kappa(l_i)}\,\bar{W}_\kappa$$

        где $m_P(l) = l$,\; $m_E(l) = \lceil l\cdot\sigma \rceil$

        \vspace{0.2cm}
        \textbf{Платёж победителя:}
        $$p_{i^*} = \phi_\kappa(\theta_j) +
                \bigl(\delta^{m_\kappa(l_j)} - \delta^{m_\kappa(l_{i^*})}\bigr)
        \bar{W}_\kappa$$

        где $j$~--- задача на втором месте,
        $\bar W_\kappa$~--- ожидаемое будущее благосостояние кластера.

        \vspace{0.2cm}
        \begin{shadowbox}
                \footnotesize
                $\phi_P(\theta_i) > \phi_E(\theta_i)$ выполняется не всегда.
                При малом $v_i$ и большом $l_i$ E-ядро может дать
                большую эффективную ценность ($c_E \lceil l_i\sigma\rceil
                < c_P l_i$ при $\sigma < c_P/c_E$).
                Оптимальная стратегия нетривиальна.
        \end{shadowbox}
\end{frame}
\addtocounter{framenumber}{-1}


\begin{frame}{Extra. Маппинг модель $\to$ реализация (VCG)}

        \begin{columns}[T]

                $m_\kappa(l_i)$. Короткие задачи делят $\phi$ и
                попадают на P в порядке FIFO внутри DSQ. Длинные
                задачи через возрастание $m_E = \lceil l\sigma \rceil$
                смещаются на E. Патовой ситуации нет. Стратегический

                \texttt{cgroup} веса.
        \end{shadowbox}

\begin{frame}{Extra. Калибровка $\sigma$, $c_P$, $c_E$ и единицы}
        \scriptsize
        \begin{shadowbox}
                \textbf{$\sigma = \eta_P / \eta_E$ из \texttt{cpu\_capacity}}\\
                Userspace-часть \texttt{scx\_A1349.c} читает
                \texttt{/sys/devices/system/cpu/cpuN/cpu\_capacity},
                $\eta_P = \max_{cpu}$, $\eta_E = \min_{cpu}$, передаёт
                в BPF через \texttt{global\_data}. На тестовой
                платформе Core Ultra 7 265K значения ядра:
                $\eta_P = 1024$, $\eta_E = 800$, отсюда
                $\sigma \approx 1{,}28$. P-кластер классифицируется
                по порогу \texttt{P\_CAP\_PCT = 90}.
        \end{shadowbox}

        \vspace{0.1cm}
        \begin{shadowbox2}
                \textbf{$c_P$, $c_E$ физически}\\
                $c_\kappa$ — удельная стоимость кванта на кластере
                $\kappa$ в единицах $w$. По умолчанию $c_P = 1024$.
                Если $c_E$ не задан, выводится $c_E = c_P \cdot
                \eta_E / \eta_P$, то есть $\gamma = c_P/c_E = \sigma$.
                Это удерживает $\phi_P \approx \phi_E$ номинально,
                смещает выбор за счёт $m_E$.
        \end{shadowbox2}

        \vspace{0.1cm}
        \begin{shadowbox}
                \textbf{Единицы}\\
                $v_i = w_i = $ \texttt{p->scx.weight}, безразмерный,
                $1$\,\ldots\,$10000$, по умолчанию $100$.\\
                $l_i$ хранится как \texttt{len\_est\_ns} (нс),
                в кванты $l_q = \texttt{len\_ns} / \texttt{SLICE\_P}$,
                \texttt{SLICE\_P} $= 20$\,мс.\\
                $B_i^{\max} = w_i \cdot \texttt{BUDGET\_MUL}$.
        \end{shadowbox}
\end{frame}

\begin{frame}{Extra. Валидация scx\_EEVDF как эталона}
        \footnotesize
        \textbf{Зачем.} \texttt{scx\_EEVDF} это эталон накладных
        расходов sched\_ext. Корректность сравнения зависит от того,
        насколько точно он воспроизводит штатный EEVDF.

        \vspace{0.1cm}
        \textbf{Воспроизведено}
        \begin{itemize}\setlength{\itemsep}{0pt}
                \item vruntime и eligible deadline
                \item lifecycle ops sched\_ext (\texttt{enqueue},
                        \texttt{dispatch}, \texttt{running} и т.д.)
                \item выбор eligible-задачи с минимальным дедлайном
        \end{itemize}

        \vspace{0.05cm}
        \textbf{Нельзя точно сохранить}
        \begin{itemize}\setlength{\itemsep}{0pt}
                \item PELT и wakeup affinity (закрытые fair-hooks)
                \item штатные балансировка и NUMA-миграции
                \item cgroups bandwidth control (частично)
        \end{itemize}

        \vspace{0.1cm}
        \begin{shadowbox}
                \scriptsize
                \textbf{Эмпирическая проверка.}
                CPU-bound (hackbench): штатный $1{,}896$\,с vs
                \texttt{scx\_EEVDF} $1{,}850$\,с (лёгкая); $9{,}729$
                vs $8{,}710$ (умеренная). Эквивалентность алгоритма
                подтверждена.

                \vspace{0.05cm}
                I/O-bound (sysbench): $4255 \to 3591$ TPS
                ($-15{,}6$\,\%), $16631 \to 12530$ TPS

\begin{frame}{Extra. Механизм выигрыша schbench p99.9}
        \footnotesize
        Трасса одного запроса schbench: \texttt{sleep} $\to$
        \texttt{matrix\_mult} $\to$ \texttt{sleep}. Метрика
        \textit{request latency} измеряет фазу matmul, она доминирует
        в хвосте p99.9.

        \vspace{0.3em}
        \begin{shadowbox2}
                \textbf{Штатный EEVDF при стрессе.}
                Фаза matmul конкурирует со всеми CPU-bound задачами
                за P-ядра единым virtual time. При $Q \gg K_P$
                лаг отдельных запросов растёт линейно по числу
                впереди стоящих, в хвосте $\Rightarrow$ p99.9
                $\approx 1{,}69$\,с.
        \end{shadowbox2}

        \vspace{0.3em}
        \begin{shadowbox2}
                \textbf{\texttt{scx\_A1349}.}
                Для matmul-фазы $l_i$ велико, $\phi_P(\theta_i) = v_i
                - c_P l_i$ остаётся выше $\phi_E(\theta_i) = v_i -
                c_E\lceil l_i\sigma\rceil$, $\arg\max_\kappa
                \phi_\kappa \to$ \texttt{AUCTION\_DSQ\_P}. Фоновые
                задачи с малым $v_i$ через тот же argmax уходят в
                \texttt{AUCTION\_DSQ\_E}. P-кластер обслуживает
                только matmul-фазы, очередь короткая,
                p99.9 $= 24{,}5$\,мс.
        \end{shadowbox2}

        \vspace{0.3em}
        \begin{shadowbox}
                \scriptsize
                \begin{tabular}{l|l}
                        EEVDF & \texttt{P:[mm|mm|bg|bg|mm|bg|\ldots]} один поток vtime\\
                        A1349 & \texttt{P:[mm|mm|mm]\quad E:[bg|bg|bg]} разделение по $\phi$
                \end{tabular}
        \end{shadowbox}
\end{frame}

\begin{frame}{Extra. \texttt{DSQ\_STARVED} и защита от голодания}
        \scriptsize
        \texttt{AUCTION\_DSQ\_STARVED} (id $= 3$) — очередь задач, не
        прошедших аукцион ($p_i > B_i^t$). Низкий приоритет,
        обслуживается после \texttt{DSQ\_P} и \texttt{DSQ\_E}.

        \vspace{0.2em}
        \begin{shadowbox2}
                \textbf{Пополнение бюджета.}
                За время сна $\Delta t$:
                $\Delta B = \min(\Delta t, \text{IDLE\_CAP}) \cdot w_i /
                \text{REPLENISH\_DIV}$, до потолка $B_i^{\max} = w_i
                \cdot \text{BUDGET\_MUL}$. Задача в \texttt{STARVED}
                накапливает бюджет до победы в следующем аукционе.
        \end{shadowbox2}

        \vspace{0.2em}
        \begin{shadowbox2}
                \textbf{Гарантия прогресса.}
                \texttt{STARVED} обслуживается FIFO по $\phi$-vtime,
                eventual progress даже при пустом бюджете. Голодание
                невозможно: либо победа после пополнения, либо
                обслуживание при простое P/E-кластеров.
        \end{shadowbox2}

        \vspace{0.2em}
        \begin{shadowbox}
                \textbf{Инверсия приоритета.}
                Отдельный PI-обработчик в BPF не реализован. Защита
                наследуется от ядерного futex PI через \texttt{rt\_mutex},
                sched\_ext видит повышенный \texttt{p->prio} и
                пересчитывает $w_i$ при следующем \texttt{enqueue}.
        \end{shadowbox}
\end{frame}
\addtocounter{framenumber}{-1}


\begin{frame}{Extra. Кэш-локальность, миграция и перегрузка P}
        \scriptsize
        \begin{shadowbox2}
                \textbf{Кэш-локальность в \texttt{select\_cpu}.}
                Быстрый путь: \texttt{prev\_cpu} без сканирования, если
                простаивает в нужном кластере. Полный путь:
                \texttt{bpf\_for} ищет свободное P-ядро. Длинные
                задачи ($l \geq \text{SLICE\_P}$) уходят в per-CPU
                sticky DSQ \texttt{PERCPU\_BASE}+cpu, сохраняя L1/L2.
        \end{shadowbox2}

        \vspace{0.2em}
        \begin{shadowbox2}
                \textbf{Стоимость миграции.}
                Явного члена нет, входит косвенно через $\sigma$:
                перенос $P\to E$ удлиняет контракт до $\lceil
                l_i\sigma\rceil$, снижает $\phi_E$ и штрафует
                миграцию для крупных $l_i$.
        \end{shadowbox2}

        \vspace{0.2em}
        \begin{shadowbox}
                \textbf{Перегрузка P-кластера:}
                $Q_P \geq K_P$ и $Q_P \cdot K_E > Q_E \cdot K_P$.
                Эквивалентно $Q_P/K_P > Q_E/K_E$ — нормированная
                глубина очереди P выше E. Перекрёстное умножение
                вместо деления убирает 64-битный \texttt{div} на
                быстром пути.

                \vspace{0.15em}
                Активируется для коротких задач ($\hat l_{\text{ns}}
                < \text{SLICE\_P}$): даже с $\sigma$-замедлением
                $c_E\lceil l_i\sigma\rceil$ меньше ожидания в очереди
                P. Trade-off: throughput при перегрузке vs строгая
                P-affinity.
        \end{shadowbox}
\end{frame}
\addtocounter{framenumber}{-1}

                        интервал
                \end{shadowbox}

                        \scriptsize
                        Системный бенчмарк, не A/B на людях:
                        коэффициент вариации $\mathrm{CV} = \sigma /
                        \mu$ мал ($\approx 0{,}2$ - $2\%$ по
                        большинству метрик)\\[0.15em]
                        Полуширины 95\% интервалов укладываются в
                        единицы процентов от среднего\\[0.15em]
                        Опора на ЦПТ для выборочного среднего, не
                        на нормальность сырых наблюдений
                \end{shadowbox2}

                        scx\_A1349 vs EEVDF на стрессе:\\
                        schbench RPS $+51\%$, p99.9 wakeup $\approx

                        Шапиро-Уилка при $N = 6$ малоинформативен,

                \end{shadowbox2}

\begin{frame}{Extra. Ограничения eBPF verifier}
        \footnotesize
        \begin{columns}[T]
                \column{0.5\textwidth}
                \begin{shadowbox}
                        \textbf{Ограничения BPF verifier}

                        \scriptsize
                        Циклы только с доказуемой границей
                        (\texttt{bpf\_loop}, \texttt{bpf\_for},
                        развёртка на этапе компиляции)\\[0.15em]
                        Рекурсия запрещена\\[0.15em]
                        Стек программы ограничен 512 байтами\\[0.15em]
                        Лимит инструкций после JIT (порядка
                        $10^6$)\\[0.15em]
                        Динамическая память общего назначения
                        недоступна, состояние только через BPF
                        maps\\[0.15em]
                        Арифметика с плавающей точкой запрещена
                \end{shadowbox}

                \column{0.5\textwidth}
                \begin{shadowbox2}
                        \textbf{Как алгоритм укладывается в эти границы}

                        \scriptsize
                        $\arg\max_\kappa \phi_\kappa$ берётся по двум
                        кластерам $\{P, E\}$ (две ветви), не по
                        $\mathcal{N}^t$\\[0.15em]
                        $\bar W_\kappa$ как EWMA реализованного
                        $\phi_\kappa$, а не точный counterfactual
                        $W_{-i}$: точный VCG-расчёт за $O(N^2)$
                        несовместим с лимитом инструкций\\[0.15em]
                        $l_i$ оценивается через EWMA
                        \texttt{len\_est\_ns}\\[0.15em]
                        Состояние задач в
                        \texttt{BPF\_MAP\_TYPE\_TASK\_STORAGE},
                        табличные данные в
                        \texttt{BPF\_MAP\_TYPE\_ARRAY}\\[0.15em]
                        Бюджет, $\phi$ и платежи в фиксированной
                        целой арифметике: \texttt{PHI\_BIAS} $= 2^{62}$,
                        \texttt{DELTA\_SCALE} $= 2^{20}$
                \end{shadowbox2}
        \end{columns}

        \vspace{0.15cm}
        \begin{shadowbox}
                \scriptsize
                Это не дефекты реализации, а проектные ограничения
                eBPF. Математическая модель сохраняет VCG-структуру
                и инвариант бюджета, упрощения относятся к оценке
                $W_{-i}$ и $l_i$.
        \end{shadowbox}
\end{frame}
