\begin{frame}{Модель системы и ключевая идея}
        \footnotesize
        \begin{columns}[T]
                \column{0.5\textwidth}
                \textbf{Аппаратная платформа}
                \begin{itemize}
                        \item Ядра: $\mathcal{P} = \{P_1,\ldots,P_{K_P}\}$,\;
                                $\mathcal{E} = \{E_1,\ldots,E_{K_E}\}$
                        \item Дискретное время $t = 1,2,3,\ldots$
                        \item Замедление на E-ядре:
                                $\sigma = \eta_P/\eta_E > 1$
                        \item Общий ISA (любая задача может быть
                                выполнена на любом ядре)
                \end{itemize}

                \column{0.5\textwidth}
                \textbf{Задачи как агенты}
                \begin{itemize}
                        \item Тип $\theta_i = (v_i, l_i)$: ценность $v_i$,
                                длина $l_i$
                        \item Полезность как маргинальный вклад
                                $U_i = W(s^t) - W_{-i}(s^t) \geq 0$
                        \item Бюджет $B_i(\chi_i) > 0$ по классу
                                обслуживания $\chi_i$:
                \end{itemize}
                \vspace{0.2em}
                \hspace{2em}{\footnotesize
                        \begin{tabular}{@{}l@{}}
                                реального времени \\
                                интерактивный \\
                                пакетный \\
                                фоновый
                \end{tabular}}
        \end{columns}

        \vspace{0.5em}
        \begin{shadowbox4}
                \scriptsize Временные кванты P/E\nobreakdash-ядер
                процессора~-- \textbf{недолговечный ресурс}
                (неиспользованный квант теряется безвозвратно). Их
                распределение моделируется динамическим аукционом VCG
                с задачами\nobreakdash-агентами типа
                $\theta_i = (v_i, l_i)$, максимизирующим социальное
                благосостояние
        \end{shadowbox4}
\end{frame}


\begin{frame}{MDP-модель и механизм VCG}
        \footnotesize
        \begin{columns}[T]
                \column{0.5\textwidth}
                \textbf{Состояние} $s^t =
                (\mathbf{x}^t,\,\mathbf{b}^t,\,\Theta^t,\,\mathcal{N}^t)$
                {\scriptsize
                \begin{itemize}\setlength{\itemsep}{0pt}
                        \item $\mathbf{x}^t$~-- остаточные длительности
                                контрактов
                        \item $\mathbf{b}^t$~-- остаточные бюджеты
                        \item $\Theta^t$~-- типы задач,
                                $\mathcal{N}^t$~-- активные задачи
                \end{itemize}}

                \vspace{0.2em}
                \textbf{Действие} $a_i^t = (a_{i,P}, a_{i,E})
                \in \{0,1\}^2$, $a_{i,P} + a_{i,E} \leq 1$
                $$\sum_i a_{i,P} \leq K_P^t,\quad
                \sum_i a_{i,E} \leq K_E^t$$
                {\scriptsize ($K_\kappa^t$~-- число свободных ядер
                кластера $\kappa$ в момент $t$)}

                \vspace{0.1em}
                \textbf{Эффективная ценность} ($\kappa \in \{P, E\}$)
                $$\phi_\kappa(\theta_i) = v_i - c_\kappa\, m_\kappa(l_i)$$
                {\scriptsize $m_P(l) = l$,\; $m_E(l) = \lceil l\sigma \rceil$}

                \column{0.5\textwidth}
                \textbf{Социальное благосостояние} (Беллман,
                $\delta \in (0,1)$):
                $$W(s^t) = \max_{a^t} \Bigl[\,
                {\textstyle\sum_i} \phi^{a}_i
                + \delta\, \mathbb{E}[W(s^{t+1})]\Bigr]$$
                {\scriptsize $\phi^{a}_i = a_{i,P}\phi_P(\theta_i)
                + a_{i,E}\phi_E(\theta_i)$}

                \vspace{4em}
                \textbf{Платёж VCG}
                $$p_i^t = W_{-i}(s^t) - \bigl(W(s^t) -
                \phi_\kappa(\theta_i)\bigr)$$

        \end{columns}
\end{frame}


\begin{frame}{Реализация}
        \footnotesize
        \textbf{Решение аукциона.} Задача направляется в одну из трёх
        кластерных DSQ по $\kappa^* = \arg\max_\kappa \phi_\kappa(\theta_i)$

        \vspace{0.3em}
        \begin{columns}[T]
                \column{0.33\textwidth}
                \begin{shadowbox2}
                        \small\textbf{DSQ\_P}

                        \vspace{0.2em}
                        \footnotesize
                        $\phi_P \geq \phi_E$

                        \vspace{0.1em}
                        P-ядро выгоднее по эффективной ценности
                \end{shadowbox2}

                \column{0.33\textwidth}
                \begin{shadowbox2}
                        \small\textbf{DSQ\_E}

                        \vspace{0.2em}
                        \footnotesize
                        $\phi_E > \phi_P$ либо короткая задача
                        при перегрузке P-кластера

                        \vspace{0.1em}
                        {\scriptsize $Q_P \geq K_P$ и $Q_P K_E > Q_E K_P$}
                \end{shadowbox2}

                \column{0.33\textwidth}
                \begin{shadowbox2}
                        \small\textbf{DSQ\_Starved}

                        \vspace{0.2em}
                        \footnotesize
                        $B_i^t < B_i^{\max} / 10$

                        \vspace{0.1em}
                        Низший приоритет, обслуживается после пополнения
                \end{shadowbox2}
        \end{columns}

        \vspace{0.6em}
        \footnotesize\textbf{Ослабления модели в реализации}

        \vspace{0.2em}
        \begin{columns}[T]
                \column{0.5\textwidth}
                {\footnotesize
                \begin{itemize}\setlength{\itemsep}{0pt}
                        \item $v_i$ наблюдается через вес задачи
                                \texttt{p->scx.weight}
                                (\texttt{nice}, \texttt{cgroup})
                        \item $W_{-i}(s^t) \to$ кластерная EWMA
                                $\bar W_\kappa$ по реализованным
                                $|\phi|$
                        \item $l_i \to$ EWMA по истории исполнения
                \end{itemize}}

                \column{0.5\textwidth}
                {\footnotesize
                \begin{itemize}\setlength{\itemsep}{0pt}
                        \item Контракт~-- горизонт планирования,
                                не атомарная единица
                        \item Бюджет~-- корзина токенов с
                                пополнением во сне
                                ($B_i^{\max} = w_i\cdot\texttt{BUDGET\_MUL}$,
                                $\Delta B_i \propto w_i$)
                \end{itemize}}
        \end{columns}
\end{frame}

\begin{frame}{Выводы}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{0.4em}
                \item На Intel Core Ultra 7 265K (8P + 12E) планировщик
                        \texttt{scx\_A1349} превосходит штатный EEVDF
                        на hackbench при трёх уровнях нагрузки на
                        $4{,}5$\,/\,$8{,}5$\,/\,$1{,}4\,\%$, а также
                        лидирует по средней пропускной способности
                        schbench при умеренной и стресс-нагрузке
                        на $51{,}3$\,/\,$55{,}7\,\%$. При стресс-нагрузке
                        задержка обслуживания запроса p99.9
                        составляет $24{,}5$\,мс против
                        $\approx 1{,}7$\,с у штатного EEVDF
                \item Определена область неприменимости. На sysbench
                        \texttt{scx\_A1349} уступает штатному EEVDF на
                        $1{,}6$\,/\,$0{,}8$\,/\,$7{,}4\,\%$. Все
                        \texttt{sched\_ext}-планировщики отстают,
                        причём \texttt{scx\_EEVDF} без учёта
                        гетерогенности~-- сильнее (до $39{,}7\,\%$),
                        что указывает на накладные расходы фреймворка
                        при высоком темпе пробуждений на IPC
                \item Атрибуция эффектов конкретным факторам

                        очереди) остаётся гипотетической, разделение

\begin{frame}{Extra}
        \textbf{Пример: $K_P = K_E = 1$, одно свободное ядро типа $\kappa$}

        \vspace{0.2cm}
        \textbf{Победитель:}
        $$i^* = \arg\max_{i \in \mathcal{N}^t,\; B_i^t > 0}\;
        \phi_\kappa(\theta_i) + \delta^{m_\kappa(l_i)}\,\bar{W}_\kappa$$

        где $m_P(l) = l$,\; $m_E(l) = \lceil l\cdot\sigma \rceil$

        \vspace{0.2cm}
        \textbf{Платёж победителя:}
        $$p_{i^*} = \phi_\kappa(\theta_j) +
                \bigl(\delta^{m_\kappa(l_j)} - \delta^{m_\kappa(l_{i^*})}\bigr)
        \bar{W}_\kappa$$

        где $j$~--- задача на втором месте,
        $\bar W_\kappa$~--- ожидаемое будущее благосостояние кластера.

        \vspace{0.2cm}
        \begin{shadowbox}
                \footnotesize
                $\phi_P(\theta_i) > \phi_E(\theta_i)$ выполняется не всегда.
                При малом $v_i$ и большом $l_i$ E-ядро может дать
                большую эффективную ценность ($c_E \lceil l_i\sigma\rceil
                < c_P l_i$ при $\sigma < c_P/c_E$).
                Оптимальная стратегия нетривиальна.
        \end{shadowbox}
\end{frame}
