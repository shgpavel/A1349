\begin{frame}{Ключевая идея и задачи}
        \footnotesize
        \begin{shadowbox4}
                \footnotesize Временные кванты P/E\nobreakdash-ядер
                процессора представляют собой \textbf{недолговечный
                ресурс} (неиспользованный квант теряется
                безвозвратно). Их распределение моделируется
                динамическим аукционом VCG с
                задачами\nobreakdash-агентами типа
                $\theta_i = (v_i, l_i)$, максимизирующим
                социальное благосостояние
        \end{shadowbox4}

        \vspace{0.5em}
        \textbf{Задачи работы}
        \begin{enumerate}\setlength{\itemsep}{0.3em}
                \item Проанализировать и определить наиболее
                        перспективный подход к реализации планировщика
                \item Построить математическую модель, учитывающую
                        различную вычислительную мощность ядер процессора
                \item Реализовать разработанный алгоритм и
                        собрать воспроизводимый набор бенчмарков
                \item Провести эксперимент на гетерогенном оборудовании
                        и оценить результаты в сравнении с базовым
                        планировщиком Linux EEVDF
        \end{enumerate}
\end{frame}

\begin{frame}{Модель системы}
        \begin{columns}[T]
                \column{0.5\textwidth}
                \textbf{Аппаратная платформа}
                \begin{itemize}
                        \item Ядра: $\mathcal{P} = \{P_1,\ldots,P_{K_P}\}$,\;
                                $\mathcal{E} = \{E_1,\ldots,E_{K_E}\}$
                        \item Дискретное время $t = 1,2,3,\ldots$
                        \item Замедление на E-ядре:
                                $\sigma = \eta_P/\eta_E > 1$
                        \item Общий ISA (любая задача может быть
                                выполнена на любом ядре)
                \end{itemize}

                \column{0.5\textwidth}
                \textbf{Задачи как агенты}
                \begin{itemize}
                        \item Тип $\theta_i = (v_i, l_i)$ (оценка и длина)
                        \item Полезность как маргинальный вклад
                                $U_i = W(s^t) - W_{-i}(s^t) \geq 0$
                        \item Бюджет $B_i > 0$ по классу обслуживания:
                \end{itemize}
                \vspace{0.2em}
                {\footnotesize
                        \begin{tabular}{@{}ll@{}}
                                Реального времени & $B_{RT}$ \\
                                Интерактивный     & $B_{int}$ \\
                                Пакетный          & $B_{batch}$ \\
                                Фоновый           & $B_{bg}$
                \end{tabular}}
        \end{columns}

        \vspace{0.4em}
        \begin{shadowbox}
                \footnotesize На E-ядре задача длиной $l_i$ исполняется за
                $m_E(l_i) = \lceil l_i \cdot \sigma \rceil$ квантов.
                Ценность $v_i$ начисляется только при полном выполнении
                контракта (допущение атомарности контракта).
        \end{shadowbox}
\end{frame}


\begin{frame}{MDP модель}
        \begin{columns}[T]
                \column{0.5\textwidth}
                \textbf{Состояние} $s^t =
                (\mathbf{x}^t,\,\mathbf{b}^t,\,\Theta^t,\,\mathcal{N}^t)$
                \begin{itemize}
                        \item $\mathbf{x}^t$~-- остаточные длительности
                                контрактов на ядрах
                        \item $\mathbf{b}^t$~-- остаточные бюджеты
                                активных задач
                        \item $\Theta^t$~-- типы текущих задач
                        \item $\mathcal{N}^t$~-- множество активных задач
                \end{itemize}

                \vspace{0.3em}
                \textbf{Действие}  $a_i^t = (a_{i,P}^t, a_{i,E}^t)
                \in \{0,1\}^2$
                $$a_{i,P}^t + a_{i,E}^t \leq 1, \quad \forall i$$
                $$\sum_i a_{i,P}^t \leq K_P^t, \quad
                \sum_i a_{i,E}^t \leq K_E^t$$

                \column{0.5\textwidth}
                \textbf{Функция перехода}
                $s^{t+1} = G(s^t, a^t, \mathcal{N}^{t+1})$
                $$x_k^{t+1} = \begin{cases}
                        m_\kappa(l_i) - 1, & \text{ядро $k$ занято $i$}\\
                        \max(x_k^t - 1,\,0), & \text{иначе}
                \end{cases}$$
                $$b_i^{t+1} = b_i^t - p_i^t$$
                {\footnotesize $m_P(l) = l$,\;
                $m_E(l) = \lceil l\sigma \rceil$~-- длина контракта}

                \vspace{0.2em}
                \textbf{Социальное благосостояние}~-- уравнение
                Беллмана, $\delta \in (0,1)$~-- дисконт:
                $$W(s^t) = \max_{a^t} \Bigl[\,
                {\textstyle\sum_i} \phi^{a}_i
                + \delta\, \mathbb{E}[W(s^{t+1})]\Bigr]$$
                {\footnotesize $\phi^{a}_i = a_{i,P}\phi_P(\theta_i)
                + a_{i,E}\phi_E(\theta_i)$ (см. след. слайд)}
        \end{columns}
\end{frame}


\begin{frame}{Механизм VCG}
        \textbf{Эффективная ценность} задачи $i$ на ядре типа
        $\kappa \in \{P, E\}$
        $$\phi_P(\theta_i) = v_i - c_P \cdot l_i, \quad
        \phi_E(\theta_i) = v_i - c_E \cdot \lceil l_i \sigma \rceil$$

        \vspace{0.1em}
        \textbf{Оптимальное распределение} (максимизация $W$)
        $$a^t = \arg\max_{a} \sum_i \bigl[a_{i,P}\,\phi_P(\theta_i)
        + a_{i,E}\,\phi_E(\theta_i)\bigr]
        + \delta\, \mathbb{E}[W(s^{t+1})]$$
        {\footnotesize $\delta \in [0,1)$~-- дисконт будущего благосостояния}

        \vspace{0.1em}
        \begin{columns}[T]
                \column{0.52\textwidth}
                \textbf{Платёж VCG} задачи $i$
                $$p_i^t = W_{-i}(s^t) - \bigl(W(s^t) -
                \phi_\kappa(\theta_i)\bigr)$$

                \vspace{0.1em}
                {\footnotesize $W(s^t),\,W_{-i}(s^t)$~-- оптимальное
                социальное благосостояние с~$i$ и без неё. Задача $i$
                платит ровно то, что теряют остальные агенты из-за её
                присутствия (pivot rule).}

                \column{0.48\textwidth}
                \begin{shadowbox}
                        \footnotesize
                        \textbf{Свойства механизма}
                        \centering
                        \begin{itemize}
                                \item Индивидуальная рациональность
                                        $u_i \geq 0$
                                \item Бюджетное ограничение
                                        $p_i \leq B_i^t$
                                \item Операционный антистимул
                                        к завышению $v_i$
                                        (платёж $\to$ бюджет)
                        \end{itemize}
                \end{shadowbox}
        \end{columns}
\end{frame}


\begin{frame}{Маршрутизация задач}
        \textbf{Решение аукциона}~-- задача направляется на кластер
        $\kappa^* = \arg\max_\kappa \phi_\kappa(\theta_i)$
        {\footnotesize (пересматривается при пробуждении задачи)}

        \vspace{0.3em}
        \begin{columns}[T]
                \column{0.33\textwidth}
                \begin{shadowbox2}
                        \small\textbf{DSQ\_P}

                        \vspace{0.3em}
                        \footnotesize
                        $\phi_P \geq \phi_E$

                        \vspace{0.2em}
                        Эффективная ценность выше
                        на P-ядре. Задача выигрывает
                        от пиковой частоты
                        производительного ядра
                \end{shadowbox2}

                \column{0.33\textwidth}
                \begin{shadowbox2}
                        \small\textbf{DSQ\_E}

                        \vspace{0.3em}
                        \footnotesize
                        $\phi_E > \phi_P$ либо
                        асимметричное перенаправление
                        коротких задач из
                        перегруженного P-кластера

                        \vspace{0.2em}
                        {\scriptsize $Q_P \geq K_P,\ \
                        Q_P K_E > Q_E K_P$}
                \end{shadowbox2}

                \column{0.33\textwidth}
                \begin{shadowbox2}
                        \small\textbf{DSQ\_Starved}

                        \vspace{0.3em}
                        \footnotesize
                        $B_i^t < B_i^{\max} / 10$

                        \vspace{0.2em}
                        Бюджет почти исчерпан.
                        Низший приоритет
                        диспетчеризации, обслуживается
                        обоими кластерами после
                        пополнения
                \end{shadowbox2}
        \end{columns}

        \vspace{0.3em}
        \begin{shadowbox}
                \footnotesize \textbf{Бюджет~-- корзина токенов.}
                $B_i^{\max} = w_i \cdot \texttt{BUDGET\_MUL}$; пополнение
                за время простоя
                $\Delta B_i = \min(\Delta t_{\text{idle}},\, T_{\max})
                \cdot w_i / D_{\text{rep}}$ обеспечивает устойчивое
                обслуживание долгоживущих задач
        \end{shadowbox}
\end{frame}

\begin{frame}{Выводы}
        \begin{itemize}\setlength{\itemsep}{0.35em}
                \item Распределение слотов P/E\nobreakdash-ядер
                        формализовано как динамический аукцион VCG:
                        состояние, действие, эффективная ценность
                        $\phi_\kappa$, платёж по pivot\nobreakdash-правилу.
                        Обоснованы стимульная совместимость в
                        пределах класса, индивидуальная рациональность
                        и бюджетное ограничение
                \item Механизм реализован как BPF\nobreakdash-планировщик
                        \texttt{scx\_A1349} поверх \texttt{sched\_ext}:
                        три кластерные DSQ, маршрутизация
                        $\kappa^* = \arg\max_\kappa \phi_\kappa$,
                        бюджет~-- корзина токенов
                \item На Intel Core Ultra 7 265K (8P+12E)
                        \texttt{scx\_A1349} быстрее штатного EEVDF на
                        hackbench при всех уровнях нагрузки
                        ($+4{,}5$\,/\,$+8{,}5$\,/\,$+1{,}4\,\%$) и лидирует
                        по RPS schbench при умеренной и стрессовой
                        нагрузке ($+51$\,/\,$+56\,\%$)
                \item При стрессе хвостовая задержка запроса p99.9
                        составляет $24{,}5$\,мс против
                        $\approx 1{,}7$\,с у штатного EEVDF. На sysbench
                        все sched\_ext-планировщики уступают штатному
                        EEVDF из-за накладных расходов фреймворка
        \end{itemize}
\end{frame}

\begin{frame}{Extra}
        \textbf{Пример: $K_P = K_E = 1$, одно свободное ядро типа $\kappa$}

        \vspace{0.2cm}
        \textbf{Победитель:}
        $$i^* = \arg\max_{i \in \mathcal{N}^t,\; B_i^t > 0}\;
        \phi_\kappa(\theta_i) + \delta^{m_\kappa(l_i)}\,\bar{W}_\kappa$$

        где $m_P(l) = l$,\; $m_E(l) = \lceil l\cdot\sigma \rceil$

        \vspace{0.2cm}
        \textbf{Платёж победителя:}
        $$p_{i^*} = \phi_\kappa(\theta_j) +
                \bigl(\delta^{m_\kappa(l_j)} - \delta^{m_\kappa(l_{i^*})}\bigr)
        \bar{W}_\kappa$$

        где $j$~--- задача на втором месте,
        $\bar W_\kappa$~--- ожидаемое будущее благосостояние кластера.

        \vspace{0.2cm}
        \begin{shadowbox}
                \footnotesize
                $\phi_P(\theta_i) > \phi_E(\theta_i)$ выполняется не всегда:
                при малом $v_i$ и большом $l_i$ E-ядро может дать
                большую эффективную ценность ($c_E \cdot \lceil l_i\sigma\rceil
                < c_P \cdot l_i$ при $\sigma < \gamma$).
                Оптимальная стратегия нетривиальна.
        \end{shadowbox}
\end{frame}
